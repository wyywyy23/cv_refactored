% Setup coffee stain
\ifcoffee
    \input{random.tex}
    \input{setup/coffee.tex}
    \usepackage{transparent}
    \newcount\coffeePicNum
    \newcount\coffeeTmpNum
    \randomi 20220110\relax
    \def\coffeehbox#1{\hbox to0pt{\kern\the\coffeeTmpNum mm #1 \hss}}
    \def\coffeevbox#1{\vbox to0pt{\kern\the\coffeeTmpNum mm #1 \vss}}
    \def\coffeerotbox#1{\rotatebox[origin=c]{\the\coffeeTmpNum}{#1}}
    \makeatletter
    \def\coffeepic#1{
        \ifcase#1\transparent{\strip@pt\dimexpr \the\coffeeTmpNum pt / 300}\coffeeA%
        \or \transparent{\strip@pt\dimexpr \the\coffeeTmpNum pt / 300}\coffeeB%
        \or \transparent{.\the\coffeeTmpNum}\coffeeC%
        \or \transparent{\strip@pt\dimexpr \the\coffeeTmpNum pt / 600}\coffeeD \fi
    }
    \makeatother
    \newcommand\randomcoffee{
        \setrannum\coffeePicNum{0}{3}% coffee stain shape 0~3
        \setrannum\coffeeTmpNum{99}{99}\edef\coffeescale{.\the\coffeeTmpNum}% scale 1~99
        \setrannum\coffeeTmpNum{-50}{-0}\coffeehbox{ % x offset
            \setrannum\coffeeTmpNum{-10}{0}\coffeevbox{% y offset
                \setrannum\coffeeTmpNum{-90}{90}\coffeerotbox{ % rotation
                    \setrannum\coffeeTmpNum{10}{60}\coffeepic\coffeePicNum}% transparency
            }%
        }%
    }
\else
    \newcommand\randomcoffee{}
\fi
